cmake_minimum_required(VERSION 3.24)
project(cuda_deep_learning_kernels LANGUAGES CXX)
add_library(cuda_deep_learning_kernels src/cpu_reference.cpp)
target_include_directories(cuda_deep_learning_kernels PUBLIC include)
target_compile_features(cuda_deep_learning_kernels PUBLIC cxx_std_20)
option(ENABLE_CUDA "Build CUDA kernels" OFF)
if(ENABLE_CUDA)
  enable_language(CUDA)
  target_sources(cuda_deep_learning_kernels PRIVATE cuda/kernels.cu)
endif()
